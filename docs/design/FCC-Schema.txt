// FCC / FURRIA — Club-App · Database schema (DBML)

Enum membership_type {
  active
  youth
  honorary
}

Enum membership_status {
  active
  paused
  left
}

// SUPERSEDED by ADR-0005 (docs/adr/0005-auth-aspnet-identity-bearer-tokens.md): the Account is
// the ASP.NET Identity user, which brings its own lockout columns. What remains of this enum is a
// single `is_disabled` flag on `account` - the club's manual switch, distinct from Identity's
// automatic lockout.
Enum account_status {
  active
  disabled
}

Table person {
  id          integer   [pk, increment]
  first_name  varchar   [not null]
  last_name   varchar   [not null]
  email       varchar
  phone       varchar
  street      varchar
  zip         varchar
  city        varchar
  created_at  timestamp [not null, default: `now()`]
  updated_at  timestamp [not null, default: `now()`]

  indexes {
    (last_name, first_name)
  }
}

Table membership {
  id          integer           [pk, increment]
  person_id   integer           [not null, unique, ref: - person.id]
  type        membership_type   [not null]
  status      membership_status [not null, default: 'active']
  started_at  date              [not null]
  ended_at    date
  created_at  timestamp         [not null, default: `now()`]
  updated_at  timestamp         [not null, default: `now()`]

  indexes {
    status
  }
}

// SUPERSEDED by ADR-0005 (docs/adr/0005-auth-aspnet-identity-bearer-tokens.md). The Account is
// now the ASP.NET Identity user entity, so Identity owns the credential columns and this hand-
// designed table is not built. Differences from what shipped in B1:
//   - `username` is dropped entirely - the email address is the only login identifier.
//   - `password_hash` and `status` are replaced by Identity's own columns plus `is_disabled`.
//   - `last_login_at` is not kept; sessions live in the `refresh_token` table instead.
//   - Identity adds `account_claim`, `account_login`, `account_token`; no role tables exist.
// The domain rules are unchanged: Account is optional, 1:1 to Person, and Mitglied != Account.
Table account {
  id             integer        [pk, increment]
  person_id      integer        [not null, unique, ref: - person.id]
  username       varchar        [not null, unique]
  email          varchar        [unique]
  password_hash  varchar        [not null]
  status         account_status [not null, default: 'active']
  last_login_at  timestamp
  created_at     timestamp      [not null, default: `now()`]
  updated_at     timestamp      [not null, default: `now()`]

  indexes {
    username
  }
}

Table invitation {
  id          integer   [pk, increment]
  person_id   integer   [not null, ref: > person.id]
  token       varchar   [not null, unique]
  invited_by  integer   [not null, ref: > person.id]
  expires_at  timestamp
  accepted_at timestamp
  created_at  timestamp [not null, default: `now()`]

  indexes {
    token
    person_id
  }
}

Table group_category {
  id         integer   [pk, increment]
  name       varchar   [not null, unique]
  created_at timestamp [not null, default: `now()`]
  updated_at timestamp [not null, default: `now()`]
}

Table group {
  id           integer   [pk, increment]
  name         varchar   [not null]
  category_id  integer   [not null, ref: > group_category.id]
  description  text
  founded_year integer
  archived_at  timestamp
  created_at   timestamp [not null, default: `now()`]
  updated_at   timestamp [not null, default: `now()`]
}

Table group_membership {
  id         integer   [pk, increment]
  person_id  integer   [not null, ref: > person.id]
  group_id   integer   [not null, ref: > group.id]
  is_trainer boolean   [not null, default: false]
  created_at timestamp [not null, default: `now()`]

  indexes {
    (person_id, group_id) [unique]
    group_id
  }
}

Table permission {
  id   integer [pk, increment]
  key  varchar [not null, unique]
  name varchar [not null]
  area varchar
}

Table role {
  id         integer   [pk, increment]
  key        varchar   [not null, unique]
  name       varchar   [not null]
  parent_id  integer   [ref: > role.id]
  is_unique  boolean   [not null, default: false]
  created_at timestamp [not null, default: `now()`]
  updated_at timestamp [not null, default: `now()`]
}

Table role_permission {
  role_id       integer [not null, ref: > role.id]
  permission_id integer [not null, ref: > permission.id]

  indexes {
    (role_id, permission_id) [pk]
  }
}

Table role_assignment {
  id          integer   [pk, increment]
  person_id   integer   [not null, ref: > person.id]
  role_id     integer   [not null, ref: > role.id]
  assigned_by integer   [ref: > person.id]
  assigned_at timestamp [not null, default: `now()`]

  indexes {
    (person_id, role_id) [unique]
    role_id
  }
}
